1. KEYCLOAK LOGIN API
================================================================================

# 1.1 Direct Keycloak Login (Username / Email & Password)
# Supports both username or email in the "username" field.
curl -X POST "https://api.soulconect.com/api/auth/login" \
  -H "Content-Type: application/json" \
  -d '{
    "username": "user@example.com",
    "password": "Password@123",
    "scope": "openid profile email offline_access"
  }'

# Alias route:
curl -X POST "https://api.soulconect.com/api/login" \
  -H "Content-Type: application/json" \
  -d '{
    "username": "user@example.com",
    "password": "Password@123"
  }'

# Sample Response (200 OK):
# {
#   "success": true,
#   "message": "Login successful",

2. FORGOT PASSWORD APIS
================================================================================

# 2.1 Keycloak Forgot Password API (New Dedicated Handler: handleForgotKCPassword)
curl -X POST "https://api.soulconect.com/api/auth/forgot-password" \
  -H "Content-Type: application/json" \
  -d '{
    "email": "user@example.com",
    "sendNativeKeycloakEmail": false
  }'

# Alias routes:

  -H "Content-Type: application/json" \
  -d '{
    "email": "user@example.com"
  }'

curl -X POST "https://api.soulconect.com/api/public/forgot-kc-password" \
  -H "Content-Type: application/json" \
  -d '{
    "email": "user@example.com"
  }'

# 2.2 Original Forgot Password API (Original Untouched Handler: handleForgotPassword)

  -H "Content-Type: application/json" \
  -d '{
    "email": "user@example.com"
  }'

3. VERIFY RESET TOKEN API
================================================================================

# 3.1 Verify Token Validity via POST Body (Before user types new password)
curl -X POST "https://api.soulconect.com/api/auth/verify-reset-token" \
  -H "Content-Type: application/json" \
  -d '{
    "token": "eyJhbGciOi..."
  }'

# 3.2 Verify Token Validity via GET Query Parameter
curl -X GET "https://api.soulconect.com/api/auth/verify-reset-token?token=eyJhbGciOi..." \
  -H "Content-Type: application/json"

# Sample Response (200 OK):
# {
#   "success": true,
#   "valid": true,
#   "message": "Reset token is valid.",
#   "email": "user@example.com",
#   "userId": "c8a149bc-43d2-43bb-a299-..."
# }

# Sample Response on Expired / Tampered Token (400 Bad Request):
# {
#   "success": false,
#   "valid": false,
#   "message": "Reset link has expired. Please request a new password reset."
# }

5. UPDATE / CHANGE PASSWORD API
================================================================================

# 5.1 Authenticated User Change Password (Verifying Current Password)
curl -X POST "https://api.soulconect.com/api/auth/update-password" \
  -H "Authorization: Bearer YOUR_ACCESS_TOKEN_HERE" \
  -H "Content-Type: application/json" \
  -d '{
    "currentPassword": "OldPassword@123",
    "newPassword": "NewSecurePassword@123",
    "logoutOtherSessions": true
  }'

# 5.2 Direct Password Change with Username/Email & Current Password
curl -X POST "https://api.soulconect.com/api/auth/update-password" \
  -H "Content-Type: application/json" \
  -d '{
    "email": "user@example.com",
    "currentPassword": "OldPassword@123",
    "newPassword": "NewSecurePassword@123",
    "logoutOtherSessions": false
  }'

# 5.3 Admin / Service Update Password by Keycloak User ID
curl -X POST "https://api.soulconect.com/api/auth/update-password" \
  -H "Content-Type: application/json" \
  -d '{
    "keycloakId": "c8a149bc-43d2-43bb-a299-...",
    "newPassword": "NewPassword@123",
    "temporary": false
  }'

# Alias route:
curl -X POST "https://api.soulconect.com/api/update-password" \
  -H "Authorization: Bearer YOUR_ACCESS_TOKEN_HERE" \
  -H "Content-Type: application/json" \
  -d '{
    "currentPassword": "OldPassword@123",
    "newPassword": "NewPassword@123"
  }'

# Sample Response (200 OK):
# {
#   "success": true,
#   "message": "Password updated successfully! You can now sign in with your new password."
# }

# Sample Response on Wrong Current Password (400 Bad Request):
# {
#   "success": false,
#   "message": "Current password is incorrect. Please verify and try again."
# }

6. LOGIN SESSION WITH REFRESH TOKEN RELATED APIS
================================================================================

# 6.1 Refresh Access Token API
# Exchanges an existing refresh token for fresh access_token & refresh_token.
curl -X POST "https://api.soulconect.com/api/auth/refresh-token" \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN_HERE"
  }'

# Alias routes:
curl -X POST "https://api.soulconect.com/api/refresh-token" \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN_HERE"
  }'

curl -X POST "https://api.soulconect.com/api/auth/refresh" \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN_HERE"
  }'

# Sample Response (200 OK):
# {
#   "success": true,
#   "message": "Token refreshed successfully",
#   "tokens": {
#     "access_token": "eyJhbGciOi...",

# 6.2 Logout & Invalidate Session API
# Revokes Keycloak refresh token, closes session, and destroys Express session store.
curl -X POST "https://api.soulconect.com/api/auth/logout" \
  -H "Authorization: Bearer YOUR_ACCESS_TOKEN_HERE" \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN_HERE",
    "keycloakId": "c8a149bc-43d2-43bb-a299-...",
    "allSessions": true
  }'

# Alias route:
curl -X POST "https://api.soulconect.com/api/logout" \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken": "YOUR_REFRESH_TOKEN_HERE"
  }'

# Sample Response (200 OK):
# {
#   "success": true,
#   "message": "Logged out successfully. Login session has been invalidated.",
#   "details": {
#     "oidcRevoked": true,
#     "adminSessionsRevoked": true
#   }
# }
